def _attack_growth(claim: str, **ctx) -> str:
    """增长错误：将增长率大幅篡改。"""
    pct_match = re.search(r"(\d+\.?\d*)\s*%", claim)
    if pct_match:
        target = pct_match.group(1)
        try:
            val = float(target)
        except ValueError:
            return claim
        factor = random.uniform(2.0, 3.5)
        new_val = round(val * factor, 1)
        return claim.replace(target, str(new_val), 1)
    return claim
